\begin{frame}
  \frametitle{Overview}
  This talk is about defining \emph{recursive algorithms} in functional languages such that they are \emph{intrinsically correct \& total}
  \begin{itemize}
  \item Following the paradigm of \emph{structured recursion}, the recursion behaviour of \emph{divide-and-conquer} algorithms can be separated from their recursive steps, which can be expressed categorically as \emph{(co)algebras} for a \emph{functor}
  \item We show how functors for intrinsically correct functional algorithms can be expressed as being \emph{locally contractive} \parencite{birkedalFirstStepsSynthetic2012}
  \item these algorithms are then \emph{intrinsically total}
  \item for recursion this means \emph{termination}, \textcolor{gray}{for corecursion \emph{productivity}}.
  \end{itemize}
\end{frame}

\section{Background}

\begin{frame}[fragile]
  \frametitle{Enriched Categories \& Enrichment}
 Fix a set of objects \(C_0\) and a monoidal category \((V,⊗,𝟙)\).
\begin{definition}[Enriched category]
  \begin{itemize}
  \item for every \(X, Y \colon C_0\) the internal Hom \((\ihom{X}{Y})\colon V\)
  \item \(\dots\) \textit{I will only show parts of definitions in this talk}
  \end{itemize}
\end{definition}
\begin{itemize}
\item Denote with \(Γ X ≔ \hom{V}{𝟙}{X}; Γ \colon V → \mathrm{Set}\). % the global element functor
  Then for enriched \(C'\) there is the \emph{underlying} plain (i.e. \(\mathrm{Set}\)-enriched) category \((Γ^*C', \ihom[']{}{})\) with \(\ihom[']{X}{Y} ≔ \hom{V}{𝟙}{(\ihom{X}{Y})} = \glob{V}{𝟙}{\ihom{X}{Y}}\)
\end{itemize}
\begin{definition}[Enrichment of a category]
  \begin{itemize}
  \item An enriched Category for \(C_0\), with
  \item \(C ≅ ΓC'\), in particular \(\expi{-}\colon \hom{C}{X}{Y} ≅ \glob{V}{𝟙}{\ihom{X}{Y}}\)
  \end{itemize}
\end{definition}
\end{frame}

\begin{frame}
  \frametitle{Families enriched in Families}
  \begin{itemize}
  \item The Category of Families \((\mathrm{Set}^{A})\) is Cartesian monodial, inheriting that structure pointwise from \(\mathrm{Set}\)
  \item In particular, \(𝟙 = ∀ i ∈ A.\, 𝟙_{\mathrm{Set}}\)
  \item The external hom is \(\hom{\mathrm{Set}^{A}}{X }{Y} = ∀ i ∈ A.\, \hom{\mathrm{Set}}{X i }{Y i}\)
  \item We consider the enrichment of families in families:
    \begin{itemize}
    \item The internal hom: \((\ihom{X}{Y}) \colon \mathrm{Set}^{A}; (\ihom{X}{Y}) i = (\hom{\mathrm{Set}^{A}}{X }{Y}) i\)
    \item External \& internal homs agree: \(\expi{-}\colon \hom{C}{X}{Y} ≅ \hom{V}{𝟙}{(\ihom{X}{Y})}\).
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{A Model of Guarded Recursive Terms}
  \def\Hom{\samebox{\ihom{X}{Y}}{\alt<2->{P}{\ihom{X}{Y}}}}%
  \begin{definition}
    Given:
    \begin{itemize}
    \item \textcolor<2->{lightgray}{A Category \(C\) enriched in Monoidal Category \(E\)}
    \item An endofunctor \(▸ \colon E → E\)
    \item A nat. trans. \(\mathrm{next} \colon \hom{[E,E]}{\mathrm{Id}}{▸}\)
    \end{itemize}
    \(E\) is a model of guarded recursive terms if:
    \begin{gather*}
      ∀ H \colon \hom{E}{▸ (\Hom)}{(\Hom)}.\ ∃!\, \mathrm{Löb}H \colon \glob{E}{𝟙}{\Hom}.\,\\
      \mathrm{Löb}H = \mathrm{Löb}H; \mathrm{next}; H
    \end{gather*}
  \end{definition}
\visible<1>{%
In particular: \(\expi{\mathrm{Löb}H} \colon \hom{C}{X}{Y}\).
I.e. \(\mathrm{Löb}\) gives a guarded fixpoint for internal homs.
}
\end{frame}

\begin{frame}[fragile]
  \frametitle{\(\mathrm{PSh}(A,<)\) as a model of guarded recursive terms}
  \begin{itemize}
  \item \(▸ \colon \mathrm{PSh}(A,<) → \mathrm{PSh}(A,<) \)\\
    \(▸ P i = ∀ j < i.\, P j\)
    % `NatTrans(∂よi,P) = ∀ ( p : j < i). P p`
  \item \(\mathrm{next}\colon \hom{\mathrm{Endo}(\mathrm{PSh}(A))}{\mathrm{Id}}{▸}\)\\
    \(\mathrm{next}\, i \colon P i → ∀ (p \colon j < i) .\, P j\) given by the presheaf's restriction maps.%
    % next takes a presheaf and sends it pointwise to to its values at strictly lower indices
  \item \(\mathrm{Löb}\) given by well founded recursion:
    \(∀ H \colon \idxbox{oiBlue}{\hom{E}{▸ P}{P}}.\ ∃!\, \mathrm{Löb}H \colon \idxbox{oiOrange}{\hom{E}{𝟙}{P}}.\, \mathrm{Löb}H = \mathrm{Löb}H; \mathrm{next}; H\)\\
    % \mathrm{Löb}H = \mathrm{Löb}H; \mathrm{next} ; H
    \(\idxbox{oiBlue}{(∀ i .\, (∀ j < i.\, P j) → P i)} → \idxbox{oiOrange}{∀ i .\, P i}\)
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Change of Base of Enrichment}
Let \((M,*,I)\) and \((N,⊗,J)\) be monoidal categories.
\begin{definition}
  \begin{itemize}
  \item   A \emph{lax monoidal} functor \(F \colon M → N\) has a comparison arrow \(ε \colon \hom{N}{J}{F I}\).
  \item It \emph{preserves underlying categories} if \(\hat{ε}_X ≔ ε ; F_1– \colon \hom{\mathrm{Set}}{\ehom{M}{I}{X}}{\ehom{N}{J}{F X}}\) is a bijection.
  \end{itemize}
\end{definition}
\begin{proposition}[Change of Base]
  Let \((C',\ihom{}{})\) be an \(M\)-enriched category for \(C_0\).
  \begin{itemize}
  \item There is an \(N\)-enriched category \((F^*C',\ihom[']{}{})\) for \(C_0\) called \emph{the base-change of \(C'\) along \(F\)} with \(\ihom[']{X}{Y} ≔ F (\ihom{X}{Y})\).
  \item If \(F\) preserves underlying categories, it induces a change of base of \emph{enrichment}. For \((EC,\expi{-})\) set \((F^*(EC),\expi{-};\hat{ε})\).
  \end{itemize}
\end{proposition}
\end{frame}

\begin{frame}
  \frametitle{Families Enriched in Presheaves}
  \begin{itemize}
  \item We already saw families enriched in families \((\mathrm{Fam}(A),\ihom{}{})\)
  \item To change their base of enrichment to presheaves, we need a lax monoidal functor \(\mathrm{Fam}(A) → \mathrm{PSh}(A)\) that preserves underlying categories.
  \item Enter the \emph{cofree presheaf functor}: \(U ⊣ \mathrm{Cofree}\), \(\mathrm{Cofree}\, X i ≔ ∀ j ≤ i. \, X j\)
  \item \textcolor{gray}{lax monoidal structure \& preservation of underlying categories derives from RAPL}
  \item Then enrichment of families in presheaves is given by \((\mathrm{Cofree}^*(\mathrm{Fam}(A)),\ihom[']{}{})\) with \(\ihom[']{X}{Y} = \mathrm{Cofree}\, (\ihom{X}{Y})\)
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Enriched Functors \& Enrichment}
  \begin{definition}[Enriched functor]
      An \emph{enriched functor} between enriched categories \((C',\ihom{}{})\), \((D',\ihom[']{}{})\) for \(C_0,D_0\), and a map \(F_0 \colon \hom{\mathrm{Set}}{C_0}{D_0}\), has an internal functorial action \(F_1' \colon \hom{V}{\ihom{X}{Y}}{\ihom[']{F_0 X}{F_0 Y}'}\).
    \end{definition}

    \begin{definition}[Enrichment of a functor]
        Given categories \(C, D\) with enrichments \((\expi{-})\) and \((\expi{-}')\), an \emph{enrichment} of a functor \(F \colon C → D\) is an enriched functor for \(F_0\) such that \(F_1\) and \(F_1'\) \emph{agree}: \(∀ f : \hom{C}{X}{Y}.\ \expi{F_1 f}' = \expi{f} ; F_1'\).
    \end{definition}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Locally Contractive Functors}
  \begin{definition}
    Given:
  \begin{itemize}
  \item \(V\) a model of guarded recursive terms
  \item Categories \(C,D\) with \(V\)-enrichments \((EC,\ihom{}{},\expi{-}), (ED,\ihom[']{}{}\expi{-}')\)
  \item Functor \(F \colon C → D\)
  \item An enriched functor \(F' \colon ▸^*EC → ED\) for \(F_0\) (its \emph{contractive action})
  \end{itemize}
  \(F\) is \emph{locally contractive} if it has as enrichment the sequence of enrichments \(EC \xrightarrow{\mathrm{next}} ▸^*EC \xrightarrow{F'} ED\), i.e.
  \[∀ f \colon \hom{C}{X}{Y}.\ \expi{F_1 f}' = \expi{f} ; \mathrm{next} ; F_1'\]
\end{definition}
\end{frame}

\begin{frame}[fragile]
  \frametitle{lcf concretely for Presheaf enrichment}
    \begin{itemize}
  \item \( f \colon \hom{\mathrm{Fam}(A)}{X}{Y} \)
  \item \(
    \begin{alignedat}{3}
      &\expi{f};\mathrm{next}\,i\, *\  && ∈ ▸ (\ihom{X}{Y}) i = ∀ j < i.\, (\ihom{X}{Y}) j \\
      &\expi{f};\mathrm{next}\,i\, *\ &&= \expi{f} \mid_{<i}
    \end{alignedat}
    \)

  \item So \enquote{\(\expi{F_1 f} = \expi{f};\mathrm{next};F_1'\)} says \enquote{\(F_1'\) is a way to apply \(F\) to Homs pointwise restricted to lower indices}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Locally Contractive Functors give Hylomorphisms}
    \begin{equation*}
      \begin{tikzcd}[every label/.append style={font=\normalsize}]
        F_0I \arrow[r,dashed,"1?"]\ar[phantom, start anchor=center, end anchor=center]{dr}[description]{(1)} & F_0O \arrow[d, "a"] \\
        I \arrow[u,"c"] \arrow[r,dashed,"0?"] &  O
      \end{tikzcd}
    \end{equation*}
\begin{itemize}
\item Note that in \eqref{eq:hylo}, the functional \(H(1?) ≔ c ; 1? ; a\) has type \(\hom{V}{(\ihom{F_0 I}{F_0 O})}{(\ihom{I}{O})}\).

\item Putting them together we get {\center \(▸ (\ihom{I}{O}) \xrightarrow{F_1'} (\ihom{F_0 I}{F_0 O}) \xrightarrow{H} (\ihom{I}{O}).\)\par{}}
\item Applying \(\mathrm{Löb} \colon (▸ (\ihom{I}{O}) → (\ihom{I}{O})) → \glob{V}{𝟙}{\ihom{I}{O}}\) gives us {\center \(\expi{\mathrm{Löb} (F_1' ; H)} \colon (I → O).\)\par{}}
\end{itemize}
\end{frame}
