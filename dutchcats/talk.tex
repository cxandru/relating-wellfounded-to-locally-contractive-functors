\begin{frame}
  \frametitle{Overview}
  This talk is about defining \emph{recursive algorithms} in functional languages such that they are \emph{intrinsically correct \& total}
  \begin{itemize}
  \item Following the paradigm of structured recursion, \emph{divide-and-conquer} algorithms can be can be expressed categorically using \emph{functors}
  \item We show how functors for \emph{intrinsically correct} algorithms can be expressed as being \emph{locally contractive} \parencite{birkedalFirstStepsSynthetic2012}
  \item We show that algorithms for such functors are \emph{intrinsically total}
    \begin{itemize}
  \item for recursion this means \emph{termination}, \textcolor{gray}{for corecursion \emph{productivity}}.
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Enriched Categories \& Enrichment}
 Fix a set of objects \(C_0\) and a monoidal category \((V,⊗,𝟙)\).
\begin{definition}[Enriched category]
  \begin{itemize}
  \item for every \(X, Y \colon C_0\) the internal Hom \((\ihom{X}{Y})\colon V\)
  \item \(\dots\) \textit{I will only show parts of definitions in this talk}
  \end{itemize}
\end{definition}
\begin{itemize}
\item Denote with \(Γ \colon V → \mathrm{Set}; Γ X ≔ \hom{V}{𝟙}{X}\). % the global element functor
  Then for enriched \(C'\) there is the \emph{underlying} plain (i.e. \(\mathrm{Set}\)-enriched) category \((Γ^*C', \ihom[']{}{})\) with \(\ihom[']{X}{Y} ≔ \hom{V}{𝟙}{(\ihom{X}{Y})} = \glob{V}{𝟙}{\ihom{X}{Y}}\)
\end{itemize}
\begin{definition}[Enrichment of a category \(C\)]
  \begin{itemize}
  \item An enriched Category \((C',\ihom{}{})\) for \(C_0\), with
  \item \(C ≅ Γ^*C'\), in particular \(\expi{-}\colon \hom{C}{X}{Y} ≅ \glob{V}{𝟙}{\ihom{X}{Y}}\)
  \end{itemize}
\end{definition}
\end{frame}

\begin{frame}
  \frametitle{Families enriched in Families}
  \begin{itemize}
  \item Intrinsically correct algorithms are expressed using indexed datatypes (families)
  \item The Category of Families \((\mathrm{Set}^{A})\) is Cartesian monodial, inheriting that structure pointwise from \(\mathrm{Set}\)
  \item In particular, \(𝟙 = ∀ i ∈ A.\, 𝟙_{\mathrm{Set}}\)
  \item The external hom is \(\hom{\mathrm{Set}^{A}}{X }{Y} = ∀ i ∈ A.\, \hom{\mathrm{Set}}{X i }{Y i}\)
  \item The internal hom is \((\ihom{X}{Y}) i = (\hom{\mathrm{Set}^{A}}{X }{Y}) i\)
  \item External \& internal homs agree: \(\expi{-}\colon \hom{C}{X}{Y} ≅ \hom{V}{𝟙}{(\ihom{X}{Y})}\).
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{A Model of Guarded Recursive Terms}
  \def\Hom{\samebox{\ihom{X}{Y\,}}{\alt<2->{P}{\ihom{X}{Y\,}}}}%
  \begin{definition}
    Given:
    \begin{itemize}
    \item \textcolor<2->{lightgray}{A Category \(C\) enriched in Monoidal Category \(E\)}
    \item An endofunctor \(▸ \colon E → E\)
    \item A nat. trans. \(\mathrm{next} \colon \hom{[E,E]}{\mathrm{Id}}{▸}\)
    \end{itemize}
    \(E\) is a model of guarded recursive terms if:
    \begin{gather*}
      ∀ H \colon \hom{E}{▸ (\Hom)}{(\Hom)}.\ ∃!\, \mathrm{Löb}H \colon \glob{E}{𝟙}{\Hom}.\,\\
      \mathrm{Löb}H = \mathrm{Löb}H; \mathrm{next}; H
    \end{gather*}
  \end{definition}
\visible<1>{%
In particular: \(\expi{\mathrm{Löb}H} \colon \hom{C}{X}{Y}\).
I.e. \(\mathrm{Löb}\) gives a guarded fixpoint.
}
\end{frame}

\begin{frame}

NB: \textcolor{gray}{gray text} indicates opaque terms.

Let \(M\) and \(N\) be monoidal categories, and \(F \colon M → N\) a \textcolor{gray}{lax monoidal} functor between them.

\begin{proposition}[Change of Base]
  Let \((C',\ihom{}{})\) be an \(M\)-enriched category for \(C_0\).
  \begin{itemize}
  \item There is an \(N\)-enriched category \((F^*C',\ihom[']{}{})\) for \(C_0\) called \emph{the base-change of \(C'\) along \(F\)} with \(\ihom[']{X}{Y} ≔ F (\ihom{X}{Y})\).
  \item If \(F\) \gluedover{preserves underlying categories}{fulfills some additional requirement}, it also induces a change of base of \emph{enrichment} \((F^*C',\expi{-}')\).
  \end{itemize}
\end{proposition}
\end{frame}

\begin{frame}
  \frametitle{Families Enriched in Presheaves}
  \begin{itemize}
  \item We already saw families enriched in families \((\mathrm{Fam}(A),\ihom{}{})\)
  \item The \emph{cofree presheaf functor}: \(U ⊣ \mathrm{Cofree}\) meets the requirements to induce a change of base of enrichment  \(\mathrm{Fam}(A) → \mathrm{PSh}(A)\).
  \item \(\mathrm{Cofree}\, X i ≔ ∀ j ≤ i. \, X j\)
  \item Then enrichment of families in presheaves is given by \((\mathrm{Cofree}^*(\mathrm{Fam}(A)),\ihom[']{}{})\) with \(\ihom[']{X}{Y} = \mathrm{Cofree}\, (\ihom{X}{Y})\)
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Enriched Functors \& Enrichment}
  \begin{definition}[Enriched functor]
      An \emph{enriched functor} for \(F_0\) between \(V\)-enriched categories has an \emph{internal functorial action} \(F_1' \colon \hom{V}{\ihom{X}{Y}}{\ihom[']{F_0 X}{F_0 Y}'}\).
    \end{definition}
    \begin{definition}[Enrichment of a functor]
      An \emph{enrichment} of a functor \(F\) is an internal functorial action that agrees with its external functorial action: \(∀ f : \hom{C}{X}{Y}.\ \expi{F_1 f}' = \expi{f} ; F_1'\)
    \end{definition}

    \begin{definition}[Locally Contractive Functor]
      With \(V\) a model of guarded recursive terms, \(F\) is \emph{locally contractive} if it has a \emph{contractive action} \(F' \colon ▸^*C' → D'\) which agrees with its external action: \(∀ f \colon \hom{C}{X}{Y}.\ \expi{F_1 f}' = \expi{f} ; \mathrm{next} ; F_1'\)

    \end{definition}
\end{frame}

  \frametitle{lcf concretely for Presheaf enrichment}
    \begin{itemize}
  \item \( f \colon \hom{\mathrm{Fam}(A)}{X}{Y} \)
  \item \(
    \begin{alignedat}{3}
      &\expi{f};\mathrm{next}\,i\, *\  && ∈ ▸ (\ihom{X}{Y}) i = ∀ j < i.\, (\ihom{X}{Y}) j \\
      &\expi{f};\mathrm{next}\,i\, *\ &&= \expi{f \mid_{<i} }\,i\, *
    \end{alignedat}
    \)

  \item So \enquote{\(\expi{F_1 f} = \expi{f};\mathrm{next};F_1'\)} says \enquote{\(F\) agrees with its contractive action, which is defined on Homs pointwise restricted to lower indices}.
  \item I.e.\ \emph{no information is lost by restricting Homs pointwise to lower indices}
  \end{itemize}

\begin{itemize}
\item Note that in \eqref{eq:hylo}, the functional \(H(1?) ≔ c ; 1? ; a\) has type \(\hom{V}{(\ihom{F_0 I}{F_0 O})}{(\ihom{I}{O})}\).

\end{itemize}
