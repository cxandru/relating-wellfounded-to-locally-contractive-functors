\begin{frame}
  \frametitle{Overview}
  This talk is about defining \emph{recursive algorithms} in functional languages which are \emph{intrinsically correct \& terminating}
  \begin{itemize}
  \item Following the paradigm of \emph{structured recursion}, the recursion behaviour of \emph{divide-and-conquer} algorithms can be separated from their recursive steps, which can be expressed categorically as \emph{(co)algebras} for a \emph{functor}
  \item We show how functors for intrinsically correct functional algorithms can be expressed as being \emph{locally contractive} \parencite{birkedalFirstStepsSynthetic2012}
  \item \emph{termination} of these algorithms then \emph{holds intrinsically}.
  \end{itemize}
\end{frame}

\section{Background}

\begin{frame}[fragile]
  \frametitle{Enriched Categories \& Enrichment}
 Fix a set of objects \(C_0\) and a monoidal category \((V,⊗,𝟙)\).
\begin{definition}[Enriched category]
  \begin{itemize}
  \item for every \(X, Y \colon C_0\) the internal Hom \(\ihom{X}{Y}\);
  \item \textcolor{gray}{an internal identity morphism \(\mathrm{id}_X \colon \hom{V}{𝟙}{(\ihom{X}{X})}\) \& sequencing operation \(\hom{V}{(\ihom{X}{Y}⊗\ihom{Y}{Z})}{(\ihom{X}{Z})}\);}
  \item \textcolor{gray}{internal counterparts to unitality \& associativity coherences}
  \end{itemize}
\end{definition}
\begin{itemize}
\item Denote with \(Γ X ≔ \hom{V}{𝟙}{X}; Γ \colon V → \mathsf{Set}\). Then for enriched \(C'\) there is the \emph{underlying} plain (i.e. \(\mathsf{Set}\)-enriched) category \((Γ^*C', \ihom[']{}{})\) with \(\ihom[']{X}{Y} ≔ \hom{V}{𝟙}{(\ihom{X}{Y})} = \glob{V}{𝟙}{\ihom{X}{Y}}\)
\end{itemize}
\begin{definition}[Enrichment of a category \(C\)]
  \begin{itemize}
  \item same data \& laws as above;
  \item \(C ≅ ΓC'\), in particular \(\expi{-}\colon \hom{C}{X}{Y} ≅ \glob{V}{𝟙}{\ihom{X}{Y}}\)
  \end{itemize}
\end{definition}
\end{frame}

  \item We consider the enrichment of families in families:

    \item The agreement \(\hom{C}{X}{Y} ≅ \hom{V}{𝟙}{(\ihom{X}{Y})}\): %between external and internal homs % \glob{V}{𝟙}{\ihom{X}{Y}} (=
    \end{itemize}
  \end{itemize}
\[\hom{\mathsf{Set}^{C_0}}{X }{Y} = \prod_{c ∈ C_0}\hom{\mathsf{Set}}{X c }{Y c} ≅ \prod_{c ∈ C_0}\hom{\mathsf{Set}}{𝟙_{\mathsf{Set}}}{\hom{\mathsf{Set}}{X c }{Y c}} = \hom{\mathsf{Set}^{C_0}}{𝟙}{\ihom{X}{Y}}\]
\end{frame}

\begin{frame}[fragile]
  \frametitle{A Model of Guarded Recursive Terms}
  \def\Hom{\samebox{\ihom{X}{Y}}{\alt<2->{P}{\ihom{X}{Y}}}}%
  Given:
  \begin{itemize}
  \item \textcolor<2->{lightgray}{A Category \(C\) enriched in Monoidal Category \(E\)}
  \item An endofunctor \(▸ \colon E → E\)
  \item A nat. trans. \(\mathsf{next} \colon \hom{[E,E]}{\mathsf{Id}}{▸}\)
  \end{itemize}
   \(E\) is a model of guarded recursive terms if:
   \[ ∀ H \colon \hom{E}{▸ (\Hom)}{(\Hom)}.\ ∃!\, \mathrm{Löb}H \colon \glob{E}{𝟙}{\Hom}\text{ such that:} \]
   \begin{equation*}
\begin{tikzcd}[every label/.append style={font=\normalsize}]
(\Hom) \arrow[r, "\mathsf{next}"]\ar[phantom, start anchor=center, end anchor=center]{dr}[description]{\refstepcounter{equation}(\theequation)\label{eq:lob}} & ▸ (\Hom) \arrow[d, "H"] \\
𝟙 \arrow[u,"\mathrm{Löb}H"] \arrow[r,"\mathrm{Löb}H"] &  (\Hom)
\end{tikzcd}
\end{equation*}
\visible<1>{%
In particular: \(\expi{\mathrm{Löb}H} \colon \hom{C}{X}{Y}\).

I.e. \(\mathrm{Löb}\) gives a guarded fixpoint for internal homs.
}
\end{frame}

\begin{frame}
  \frametitle{Enrichment of a Functor}
\begin{definition}[Enrichment of a Functor]
  Given:
  \begin{itemize}
  \item Categories \(C, D\) with \(V\)-enrichments \((EC,\ihom{}{},\expi{-})\) and \((ED,\ihom[']{}{},\expi{-}')\);
  \item Functor \(F \colon C → D\)
  \end{itemize}
   An \emph{enrichment} of \(F\) in \(EC,ED\) is an internal functorial action
   \[F_1' \colon \hom{V}{\ihom{X}{Y}}{\ihom[']{F_0 X}{F_0 Y}'}\]
   \vspace{-1\baselineskip{}}
   \begin{itemize}
   \item \textcolor{gray}{obeying internal counterparts to preservation of id and comp.;}
   \item and such that \(F_1\) and \(F_1'\) \emph{agree}: (\(\expi{-}\colon \hom{C}{X}{Y} ≅ \glob{V}{𝟙}{\ihom{X}{Y}}\))
   \end{itemize}
  \[∀ X, Y : C,\, f : \hom{C}{X}{Y}.\ \expi{F_1 f}' = \expi{f} ; F_1' \]
\end{definition}  
\end{frame}

\begin{frame}[fragile]
  \frametitle{Locally Contractive Functors}
  \begin{definition}
    Given:
  \begin{itemize}
  \item \(V\) a model of guarded recursive terms
  \item Categories \(C,D\) with \(V\)-enrichments \((EC,\ihom{}{},\expi{-}), (ED,\ihom[']{}{}\expi{-}')\)
  \item Functor \(F \colon C → D\)
  \item An enriched functor \(F' \colon ▸^*EC → ED\) with \(F_0\) as object map
  \end{itemize}
    \(F\) is \emph{locally contractive} if it has as enrichment the sequence of enrichments \(EC \xrightarrow{\mathsf{next}} ▸^*EC \xrightarrow{F'} ED\), i.e. \(∀f : \hom{C}{X}{Y}.\)
\begin{equation*}
\begin{tikzcd}[every label/.append style={font=\normalsize}]
(\ihom{X}{Y}) \arrow[r, "\mathsf{next}"] & ▸ (\ihom{X}{Y}) \arrow[d, "F_1'"] \\
𝟙 \arrow[u,"\expi{f}"] \arrow[r,"\expi{F_1 f}'"] &  (\ihom[']{F_0X}{F_0Y})
\end{tikzcd}
\end{equation*}
\end{definition}
\end{frame}

\begin{frame}[fragile]
  \frametitle{lcf concretely for Presheaf enrichment}
    \begin{itemize}
  \item \( f \colon \hom{\mathsf{Fam}(A)}{X}{Y} \)
  \item \(
    \begin{alignedat}{3}
      &\expi{f};\mathsf{next} && \colon ▸ (\ihom{X}{Y}) = ∀ (j < i).\, \ihom{X}{Y} j \\
      &\expi{f};\mathsf{next}\ i\ &&= \expi{f} \mid_{<i}
    \end{alignedat}
    \)

  \item So \enquote{\(\expi{F_1 f} = \expi{f};\mathsf{next};F_1'\)} says \enquote{\(F_1'\) is a way to apply \(F\) to Homs pointwise restricted to lower indices}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Locally Contractive Functors give Hylomorphisms}
    \begin{equation*}
      \begin{tikzcd}[every label/.append style={font=\normalsize}]
        F_0I \arrow[r,dashed,"1?"]\ar[phantom, start anchor=center, end anchor=center]{dr}[description]{(1)} & F_0O \arrow[d, "a"] \\
        I \arrow[u,"c"] \arrow[r,dashed,"0?"] &  O
      \end{tikzcd}
    \end{equation*}
\begin{itemize}
\item Note that in \eqref{eq:square}, the functional \(H(h) ≔ c ; h ; a\) has type \(\hom{V}{(\ihom{F_0 I}{F_0 O})}{(\ihom{I}{O})}\).
\item Local contractivity of \(F\) gives us \(F_1' \colon \hom{V}{▸ (\ihom{I}{O})}{(\ihom{F_0 I}{F_0 O})}\).
\item Putting them together we get {\center \(▸ (\ihom{I}{O}) \xrightarrow{F_1'} (\ihom{F_0 I}{F_0 O}) \xrightarrow{H} (\ihom{I}{O}).\)\par{}}
\item Applying \(\mathrm{Löb} \colon (▸ (\ihom{I}{O}) → (\ihom{I}{O})) → \glob{V}{𝟙}{\ihom{I}{O}}\) gives us {\center \(\expi{\mathrm{Löb} (F_1' ; H)} \colon (I → O).\)\par{}}
\end{itemize}
\end{frame}

  \item We already saw families enriched in families \((\mathsf{Fam}(A),\ihom{}{})\)
  \item To change their base of enrichment to presheaves, we need a lax monoidal functor \(\mathsf{Fam}(A) → \mathsf{PSh}(A)\)
  \item Enter the \emph{cofree presheaf functor}: \(U ⊣ \mathsf{Cofree}\), \(\mathsf{Cofree}(B)(x) ≔ ∀ y ≤ x. \, B(x)\)
  \item \textcolor{gray}{lax monoidal structure derives from RAPL}
  \item Then enrichment of families in presheaves is given by \((\mathsf{Cofree}^*(\mathsf{Fam}(A)),\ihom[']{}{})\) with \(\ihom[']{X}{Y} = \mathsf{Cofree}\, (\ihom{X}{Y})\)
  \end{itemize}
\end{frame}
